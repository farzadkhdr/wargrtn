پەیوەندی لەگەڵ پشتەندی C++

ئەم سیستەمە لەگەڵ پشتەندی C++ لەڕێگەی API ڕێستفۆڵ دەپەیوەستێت. نموونەی کۆدی C++ بۆ پەیوەندیکردن لەگەڵ بنکەدراوەی MySQL:


#include <mysql.h>
#include <iostream>
#include <string>

// پەیوەندیکردن لەگەڵ بنکەدراوە
MYSQL* connectToDatabase() {
    MYSQL* conn = mysql_init(NULL);
    if (conn == NULL) {
        std::cerr << "هەڵە لە دەستپێکردنی MySQL" << std::endl;
        return NULL;
    }
    
    if (mysql_real_connect(conn, "localhost", "user", "password", 
                          "hospital_system", 0, NULL, 0) == NULL) {
        std::cerr << "هەڵە لە پەیوەندیکردن: " << mysql_error(conn) << std::endl;
        mysql_close(conn);
        return NULL;
    }
    
    std::cout << "پەیوەندی دروستکرا لەگەڵ بنکەدراوە!" << std::endl;
    return conn;
}

// هێنانەوەی نەخۆشەکان
void getPatients(MYSQL* conn) {
    std::string query = "SELECT * FROM patients ORDER BY درووستکراوە_لە DESC LIMIT 10";
    
    if (mysql_query(conn, query.c_str())) {
        std::cerr << "هەڵە لە داواکاری: " << mysql_error(conn) << std::endl;
        return;
    }
    
    MYSQL_RES* result = mysql_store_result(conn);
    if (result == NULL) {
        std::cerr << "هەڵە لە وەرگرتنی ئەنجامەکان" << std::endl;
        return;
    }
    
    int num_fields = mysql_num_fields(result);
    MYSQL_ROW row;
    
    while ((row = mysql_fetch_row(result))) {
        for (int i = 0; i < num_fields; i++) {
            std::cout << (row[i] ? row[i] : "NULL") << " | ";
        }
        std::cout << std::endl;
    }
    
    mysql_free_result(result);
}
